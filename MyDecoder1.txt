module MyDecoder1#(
parameter WIDTH = 8,
parameter OUTW = 2**WIDTH
)(
input [WIDTH-1:0] a,
output reg [OUTW-1:0] q
    );
    
     always @(*)begin
     q = {OUTW{1'b0}};
     q[a] = 1'b1;
     end   
endmodule
